solution: |
  #include <iostream>
  #include <string>

  class Resource {
  public:
      std::string name;

      Resource(std::string n) : name(n) {
          std::cout << "Created: " << name << std::endl;
      }

      ~Resource() {
          std::cout << "Destroyed: " << name << std::endl;
      }
  };

  int main() {
      std::cout << "before" << std::endl;
      {
          Resource r("widget");
          std::cout << "using widget" << std::endl;
      }
      std::cout << "after" << std::endl;
      return 0;
  }

Add a destructor to `Resource` that prints `Destroyed: <name>`. The expected output pins down exactly *when* it
should fire: right as the inner `{ }` block ends, after `"using widget"` but before `"after"` -- not at the end of
`main`, and not anywhere you'd need to call it explicitly.
