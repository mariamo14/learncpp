solution: |
  #include <iostream>

  class Point2D {
  public:
      double x;
      double y;

      Point2D(double px, double py) : x(px), y(py) {
      }
  };

  std::ostream &operator<<(std::ostream &out, const Point2D &p) {
      out << "(" << p.x << ", " << p.y << ")";
      return out;
  }

  int main() {
      double x, y;
      std::cin >> x >> y;
      Point2D p(x, y);
      std::cout << p << std::endl;
      return 0;
  }
---

Define `operator<<` so `std::cout << p` prints `Point2D p` as `(x, y)`. Write it as a free function, outside the
class -- exactly the lesson's `Vector2D` example, applied to `Point2D` -- taking the stream and the point as two
explicit parameters, and returning the stream by reference so `std::cout << p << std::endl;` keeps chaining
correctly.
