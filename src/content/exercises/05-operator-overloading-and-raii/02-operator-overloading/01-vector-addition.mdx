solution: |
  #include <iostream>

  class Point2D {
  public:
      double x;
      double y;

      Point2D(double px, double py) : x(px), y(py) {
      }

      Point2D operator+(const Point2D &other) const {
          return Point2D(x + other.x, y + other.y);
      }
  };

  int main() {
      double x1, y1, x2, y2;
      std::cin >> x1 >> y1 >> x2 >> y2;
      Point2D a(x1, y1);
      Point2D b(x2, y2);
      Point2D c = a + b;
      std::cout << "x = " << c.x << ", y = " << c.y << std::endl;
      return 0;
  }

Add `operator+` to `Point2D` so `a + b` produces a new point whose coordinates are the sum of `a`'s and `b`'s.
`main` already writes `Point2D c = a + b;` -- your only job is making that expression mean something.
