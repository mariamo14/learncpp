solution: |
  #include <iostream>

  class Point2D {
  public:
      double x;
      double y;

      Point2D(double px, double py) : x(px), y(py) {
      }

      bool operator==(const Point2D &other) const {
          return x == other.x && y == other.y;
      }
  };

  int main() {
      double x1, y1, x2, y2;
      std::cin >> x1 >> y1 >> x2 >> y2;
      Point2D a(x1, y1);
      Point2D b(x2, y2);
      std::cout << (a == b ? "Equal" : "Not equal") << std::endl;
      return 0;
  }
---

Add `operator==` so `a == b` compares two `Point2D`s member-wise, true only when both `x` and `y` match. This is
the direct C++ analogue of Module 4's `std::string ==` -- except this time you're the one defining what equality
means for the type, since `Point2D` isn't a standard type with equality already built in.
