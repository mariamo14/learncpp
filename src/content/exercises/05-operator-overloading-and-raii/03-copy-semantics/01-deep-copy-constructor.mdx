solution: |
  #include <iostream>

  class Buffer {
  public:
      int *data;
      int size;

      Buffer(int n) : data(new int[n]), size(n) {
          for (int i = 0; i < n; i++) {
              data[i] = i;
          }
      }

      Buffer(const Buffer &other) : data(new int[other.size]), size(other.size) {
          for (int i = 0; i < size; i++) {
              data[i] = other.data[i];
          }
      }

      ~Buffer() {
          delete[] data;
      }
  };

  int main() {
      int n;
      std::cin >> n;
      Buffer a(n);
      Buffer b = a;

      b.data[0] = 999;

      std::cout << "a.data[0] = " << a.data[0] << std::endl;
      std::cout << "b.data[0] = " << b.data[0] << std::endl;
      return 0;
  }
---

Add a deep-copy copy constructor to `Buffer`. `main` creates `b` as a copy of `a`, then modifies `b.data[0]` --
with a correct deep copy, that change should be completely invisible to `a`, since they now own two entirely
separate arrays with (initially) equal contents.

If `a.data[0]` also ends up `999` in your output, the copy is shallow -- both objects are still sharing one array.
