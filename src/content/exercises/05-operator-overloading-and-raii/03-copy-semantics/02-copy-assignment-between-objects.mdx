solution: |
  #include <iostream>

  class Buffer {
  public:
      int *data;
      int size;

      Buffer(int n) : data(new int[n]), size(n) {
          for (int i = 0; i < n; i++) {
              data[i] = i;
          }
      }

      Buffer(const Buffer &other) : data(new int[other.size]), size(other.size) {
          for (int i = 0; i < size; i++) {
              data[i] = other.data[i];
          }
      }

      Buffer &operator=(const Buffer &other) {
          delete[] data;
          data = new int[other.size];
          size = other.size;
          for (int i = 0; i < size; i++) {
              data[i] = other.data[i];
          }
          return *this;
      }

      ~Buffer() {
          delete[] data;
      }
  };

  int main() {
      int n1, n2;
      std::cin >> n1 >> n2;
      Buffer a(n1);
      Buffer b(n2);

      b = a;
      b.data[0] = 999;

      std::cout << "a.data[0] = " << a.data[0] << std::endl;
      std::cout << "b.size = " << b.size << std::endl;
      std::cout << "b.data[0] = " << b.data[0] << std::endl;
      return 0;
  }
---

`a` and `b` are two already-existing `Buffer`s of possibly different sizes -- `b = a;` needs to make `b` an
independent deep copy of `a`, resizing `b`'s array to match `a`'s in the process. This is genuinely different work
from the copy constructor: `b` already owns memory from its own constructor, and that memory has to be released
before `b` can take on `a`'s size and contents, or the original allocation simply leaks.

This exercise doesn't require a self-assignment guard (`a` and `b` are always distinct objects here) -- the next
exercise covers exactly that case on its own.
